class Solution {
public:
    int largestRectangleArea(vector<int>& heights) {
        stack<pair<int,int>>s;
        int n=heights.size();
        int ans=0;
        for(int i=0;i<n;i++){
            while(!s.empty() && s.top().first>heights[i]){
                ans=max(ans,s.top().first*(i-s.top().second));
                s.pop();
            }
            s.push({heights[i],i});
        }

        while(!s.empty()){
            ans=max(ans,s.top().first*(n-0)); 
            s.pop();
        }

        return ans;
    }
};
