class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        unordered_map<int,int>m;
        stack<int>s;
        for(auto e:nums2){
            while(!s.empty() && s.top()<e){
                m[s.top()]=e;
                s.pop();
            }
            s.push(e);
        }

        vector<int>ans(nums1.size(),-1);
        for(int i=0;i<nums1.size();i++){
            if(m.contains(nums1[i])>0){
                ans[i]=m[nums1[i]];
            }
        }

        return ans;
    }
};
