/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:

    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if(root==NULL)
        return NULL;
        if(root==p || root==q){
            return root ;
        }
        
        TreeNode* p1=lowestCommonAncestor(root->left,p,q);
        TreeNode* p2=lowestCommonAncestor(root->right,p,q);
        if((p1==p && p2==q) ||(p1==q && p2==p) ){
            return root;
        }

        if(p1!=q && p2!=p && p1!=p && p2!=q){
            if(p1==NULL){
                return p2;
            }else{
                return p1;
            }
        }


        if(p1==q || p2==p || p2==q|| p1==p){
            if(p1==NULL){
                return p2;
            }else{
                return p1;
            }
        }

        return NULL;


        

    }
};
