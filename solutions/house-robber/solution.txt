class Solution {
public:
    int rob(vector<int>& nums) {
        int n=nums.size();
        vector<int>dp(n,0);
        dp[0]=nums[0];
        dp[1]=nums[1];
        int ans=max(dp[0],dp[1]);
        for(int i=2;i<n;i++){
            int x=0;
            for(int j=i-2;j>=0;j--){
               x=max(x,dp[j]+nums[i]);
            }
            dp[i]=x;
            ans=max(ans,dp[i]);
        }

        return ans;

    }
};
