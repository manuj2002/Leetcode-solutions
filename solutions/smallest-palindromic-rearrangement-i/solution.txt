class Solution {
public:
    string smallestPalindrome(string s) {
        vector<int>fre(26,0);
        for(auto e:s){
            fre[e-'a']++;
        }
        string ans="";
        int c;

        for(int i=0;i<26;i++){
            if(fre[i]==0)
              continue;
              int n=0;
              if(fre[i]%2==0){
                n=fre[i]/2;
                fre[i]=(fre[i]/2);
              }else{
                 c=i;
                 n=fre[i]/2;
                 fre[i]=(fre[i]/2);
              }

              while(n--){
                ans+=('a'+i);
              }
        }

        if(c!=0){
            ans+=(c+'a');
        }

        for(int i=0;i<26;i++){
            if(fre[i]==0)
              continue;

              while(fre[i]--){
                ans+=('a'+i);
              }
        }


         return ans;
    }
};
